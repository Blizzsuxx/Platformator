cmake_minimum_required(VERSION 3.16)
project(PlatformatorConsumer LANGUAGES CXX)
if(PLATFORMATOR_SOURCE_DIR)
    set(PLATFORMATOR_BUILD_RUNNER OFF CACHE BOOL "" FORCE)
    set(PLATFORMATOR_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(PLATFORMATOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(PLATFORMATOR_INSTALL OFF CACHE BOOL "" FORCE)
    add_subdirectory("${PLATFORMATOR_SOURCE_DIR}" platformator)
else()
    find_package(Platformator CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE Platformator::platformator)
set_target_properties(consumer PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
install(TARGETS consumer RUNTIME DESTINATION bin)
